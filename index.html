/** * Field Sales Tracker — free Google Sheet backend. * * SETUP (one-time, ~5 minutes, 100% free): * 1. Go to https://sheets.google.com and create a new blank spreadsheet. * Name it "Field Sales Tracker Data". * 2. In that sheet: Extensions > Apps Script. * 3. Delete any starter code in Code.gs, paste in this whole file, and save (Ctrl/Cmd+S). * 4. Click Deploy > New deployment. * - Click the gear icon next to "Select type" and choose "Web app". * - Description: anything, e.g. "Field Sales Tracker API". * - Execute as: "Me". * - Who has access: "Anyone". * - Click Deploy, then Authorize access (choose your Google account, click * "Advanced" > "Go to project (unsafe)" > Allow — this warning is normal * for your own scripts). * 5. Copy the "Web app URL" it gives you (ends in /exec). * 6. Open the Field Sales Tracker app link, tap the gear icon (top right), * paste that URL, and tap Save. Do this on every salesperson's phone, * using the SAME url each time. * * That's it — every check-in, quote, order, and trail point will now write * into this Sheet, and the "Team activity" section on the Dashboard tab * will read live from it. Photos are saved into a Drive folder called * "FieldSalesPhotos" (created automatically) and linked from the Sheet. */ function doGet(e) { var sheetName = (e.parameter && e.parameter.sheet) || 'Checkins'; var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(sheetName); if (!sheet) { return ContentService.createTextOutput(JSON.stringify([])) .setMimeType(ContentService.MimeType.JSON); } var data = sheet.getDataRange().getValues(); var headers = data.shift() || []; var rows = data.map(function (r) { var obj = {}; headers.forEach(function (h, i) { obj[h] = r[i]; }); return obj; }); return ContentService.createTextOutput(JSON.stringify(rows)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var body = JSON.parse(e.postData.contents); var type = body.type; var sheetNames = { checkin: 'Checkins', quote: 'Quotes', order: 'Orders', trail: 'Trail' }; var sheetName = sheetNames[type]; if (!sheetName) { return ContentService.createTextOutput(JSON.stringify({ ok: false, error: 'unknown type' })) .setMimeType(ContentService.MimeType.JSON); } var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(sheetName); if (!sheet) sheet = ss.insertSheet(sheetName); var headerMap = { Checkins: ['Timestamp', 'Salesperson', 'Customer', 'Channel', 'IsNew', 'PhotoUrl', 'Lat', 'Lng', 'Minutes'], Quotes: ['Timestamp', 'Salesperson', 'Customer', 'QType', 'FileName', 'PhotoUrl'], Orders: ['Timestamp', 'Salesperson', 'Customer', 'Classes', 'Notes'], Trail: ['Timestamp', 'Salesperson', 'Lat', 'Lng', 'Label'] }; if (sheet.getLastRow() === 0) sheet.appendRow(headerMap[sheetName]); var photoUrl = ''; if (body.photo) { try { photoUrl = savePhoto_(body.photo, type + '_' + Date.now()); } catch (err) { photoUrl = ''; } } if (type === 'checkin') { sheet.appendRow([body.ts, body.sp, body.customer, body.subtype, body.isNew, photoUrl, body.lat || '', body.lng || '', body.minutes || '']); } else if (type === 'quote') { sheet.appendRow([body.ts, body.sp, body.customer, body.qtype, body.fileName, photoUrl]); } else if (type === 'order') { sheet.appendRow([body.ts, body.sp, body.customer, (body.classes || []).join(', '), body.notes]); } else if (type === 'trail') { sheet.appendRow([body.ts, body.sp, body.lat, body.lng, body.label || '']); } return ContentService.createTextOutput(JSON.stringify({ ok: true, photoUrl: photoUrl })) .setMimeType(ContentService.MimeType.JSON); } function savePhoto_(base64Data, name) { var folderName = 'FieldSalesPhotos'; var folders = DriveApp.getFoldersByName(folderName); var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName); var parts = base64Data.split(','); var meta = parts[0]; var data = parts[1]; var contentType = meta.split(':')[1].split(';')[0]; var blob = Utilities.newBlob(Utilities.base64Decode(data), contentType, name); var file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return 'https://drive.google.com/uc?id=' + file.getId(); }
